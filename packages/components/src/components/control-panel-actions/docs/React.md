## General

For general installation and configuration look at the [project documentation](https://www.npmjs.com/package/@db-ux/react-core-components).

## Usage

```tsx App.tsx
// App.tsx
import {
	DBControlPanelActions1,
	DBControlPanelActions2
} from "@db-ux/react-core-components";

const App = () => (
	<>
		<DBControlPanelActions1>ControlPanelActions1</DBControlPanelActions1>
		<DBControlPanelActions2>ControlPanelActions2</DBControlPanelActions2>
	</>
);

export default App;
```
