import {
	Description,
	Primary,
	Stories,
	Subtitle,
	Title,
	useOf
} from "@storybook/addon-docs/blocks";
import overviewLinks from "../src/overview/overview-links.json";

{/* Docs page of the generated example stories, without the properties table (it is on the Overview page). Copied into every Storybook by storybooks/_shared/storybook-config/sync.ts – edit the shared file. */}

export const OverviewLink = () => {
	const { preparedMeta } = useOf("meta");
	const segments = preparedMeta.title.split("/").slice(0, -1);
	// Parts and family members (DBTabItem, DBCustomHeading) are documented on
	// the page of their main component, see overview-links.json of
	// generate-overviews.ts. Otherwise the Overview page of the same group.
	const id =
		overviewLinks[segments.at(-1)] ??
		`${segments.join("/")}/Overview--docs`
			.toLowerCase()
			.replace(/\//g, "-")
			.replace(/\s+/g, "-");
	return (
		<p>
			Properties, usage and all examples of the component are on its{" "}
			<a href={`?path=/docs/${id}`} target="_top">
				Overview
			</a>{" "}
			page.
		</p>
	);
};

<Title />

<Subtitle />

<Description />

<OverviewLink />

<Primary />

<Stories />
